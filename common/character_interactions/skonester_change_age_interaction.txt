skonester_change_age_interaction = {
    icon = on
    category = interaction_category_uncategorized
    common_interaction = yes
    interface_priority = 11790
    
    is_highlighted = { always = yes }
    
    auto_accept = yes
    ai_will_do = -100
    use_diplomatic_range = no
    ignores_pending_interaction_block = yes
    
    is_shown = {
    	scope:actor = { has_character_flag = skonester_character_interaction_menu_open }
		scope:actor = { is_ai = no has_character_flag = bone_cheat_interactions_flag }
    }
    
    send_option = {
        flag = flag_age_6
        localization = flag_age_6
    }
    send_option = {
        flag = flag_age_16
        localization = flag_age_16
    }
    send_option = {
        flag = flag_age_25
        localization = flag_age_25
    }
    send_option = {
        flag = flag_age_42
        localization = flag_age_42
    }
    send_option = {
        flag = flag_age_64
        localization = flag_age_64
    }
    send_options_exclusive = yes

    on_accept = {
        scope:recipient = {
            switch = {
                trigger = yes
                
                scope:flag_age_6 = {
                    set_age = 6
                }
                scope:flag_age_16 = {
                    set_age = 16
                }
                scope:flag_age_25 = {
                    set_age = 25
                }
                scope:flag_age_42 = {
                    set_age = 42
                }
                scope:flag_age_64 = {
                    set_age = 64
                }
            }
        }
    }
}



debug_imprison_simple_interaction = {
	use_diplomatic_range = no
	category = interaction_category_uncategorized
	common_interaction = yes
	ignores_pending_interaction_block = yes
	icon = prison
	interface_priority = 11750

	is_shown = {
		scope:actor = { has_character_flag = skonester_character_interaction_menu_open }
		NOT = { scope:actor = scope:recipient }  # Prevents self-imprisonment
		scope:actor = { is_ai = no has_character_flag = bone_cheat_interactions_flag }
		scope:recipient = {
			is_imprisoned = no
		}
	}

	on_accept = {
		scope:actor = {
			imprison = {
				target = scope:recipient
				type = house_arrest
			}
		}
	}

	auto_accept = yes
}
