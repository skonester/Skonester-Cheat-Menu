toc_deposed_interaction = {
	category = interaction_category_uncategorized
	common_interaction = yes
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	icon = on
	desc = toc_deposed_interaction_desc
	
	interface_priority = 11740

	is_shown = {
		scope:actor = { has_character_flag = skonester_character_interaction_menu_open }
		scope:actor = { is_ai = no has_character_flag = bone_cheat_interactions_flag }
		scope:recipient = { is_landed = yes }
	}

	is_valid_showing_failures_only = {
		# A player cannot remain playable after losing every title unless landless
		# play is available. AI recipients can still be removed normally.
		trigger_if = {
			limit = { scope:recipient = { is_ai = no } }
			scope:recipient = {
				custom_tooltip = {
					text = skonester_depose_requires_landless_play_tt
					has_ep3_dlc_trigger = yes
				}
			}
		}
	}

	on_accept = {
		scope:recipient = {
			# Administrative governors must retain their noble-family title. Raw
			# `depose = yes` also removes that title and immediately game-overs a
			# player, so use the current governorship retirement path instead.
			if = {
				limit = {
					has_ep3_dlc_trigger = yes
					is_governor_or_admin_count = yes
					top_liege = { government_allows = noble_families }
				}
				if = {
					limit = {
						NOT = { any_held_title = { is_noble_family_title = yes } }
					}
					top_liege = { save_scope_as = skonester_estate_government_giver }
					create_noble_family_effect = {
						GOVERNMENT_GIVER = scope:skonester_estate_government_giver
					}
					domicile ?= { set_up_domicile_estate_effect = yes }
				}
				if = {
					limit = { any_held_title = { is_noble_family_title = yes } }
					custom_tooltip = skonester_deposed_to_estate_tt
					hidden_effect = { force_step_down_landed_titles = yes }
				}
				else = {
					# Defensive fallback if another mod prevents estate creation.
					depose_effect = { DEPOSER = scope:actor }
				}
			}
			else = {
				# Current base-game path for non-administrative rulers. With EP3 this
				# opens the continuation-as-an-adventurer flow instead of game over.
				depose_effect = { DEPOSER = scope:actor }
			}
		}
	}
	auto_accept = yes
}
