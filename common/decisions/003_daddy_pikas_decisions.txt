change_government_celestial = {
    picture = {
		reference = "gfx/interface/illustrations/decisions/cheat_tab.dds"
	}

    title = change_govt_celestial
    desc = change_govt_celestial_desc
    selection_tooltip = change_govt_tt
    confirm_text = CONFIRM_OKAY

	decision_group_type = skonester_realm_tools

    is_shown = {
        # CRITICAL FIX MISSING: You must add 'is_player = yes' here for visibility.
        NOT = { has_government = celestial_government }
        has_character_flag = daddy_pika_cheat_menu
        has_character_flag = govt_menu
    }

    effect = {
        # Prepare the target's current native primary holding before changing government.
        every_sub_realm_county = {
            title_province = {
                if = {
                    limit = { NOT = { has_holding_type = castle_holding } }
                    set_holding_type = castle_holding
                    skonester_ensure_castle_holding_base_castle_building = yes
                }
            }
        }

        # Custom mod effect to clean up landless status (requires third-party mod)

		skonester_121_prepare_government_domicile_effect = {
            NEW_DOMICILE_TYPE = east_asian_estate
            NEW_GOVERNMENT_TYPE = celestial_government
        }
        # Change player's government
		skonester_121_transition_to_celestial_government_effect = yes

        # WARNING: This mass converts holdings and is highly disruptive/game-breaking
		

        # Change vassals' government
		every_vassal_or_below = {
			skonester_121_transition_to_celestial_government_effect = yes
		}
    }
    ai_check_interval = 0
    ai_will_do = 0
}

change_government_landless_adventurer = {
    picture = { reference = "gfx/interface/illustrations/decisions/cheat_tab.dds" }

    title = change_govt_landless_adventurer
    desc = change_govt_desc
    selection_tooltip = change_govt_tt
    confirm_text = CONFIRM_OKAY

    decision_group_type = skonester_realm_tools

    is_shown = {
        has_character_flag = daddy_pika_cheat_menu
        has_character_flag = govt_menu
    }

    effect = {
        # Required scope the event reads from
        save_scope_as = adventurer

        # Determine who gets the titles — mirrors vanilla logic
        if = {
            limit = {
                player_heir ?= {
                    this != root
                    is_adult = yes
                }
            }
            player_heir = {
                save_scope_as = laamp_heir
                save_scope_as = laamp_inheritor
            }
        }
        else_if = {
            limit = { top_liege != this }
            liege = {
                save_scope_as = laamp_liege
                save_scope_as = laamp_inheritor
            }
        }
        else = {
            # No valid heir or liege — generate a throwaway inheritor character
            hidden_effect_new_object = {
                create_character = {
                    template = laamp_inheritor_template
                    culture = root.capital_county.culture
                    faith = root.capital_county.faith
                    dynasty = generate
                    location = root.capital_province
                    after_creation = {
                        save_scope_as = laamp_inheritor
                        add_character_flag = new_laamp_inheritor
                    }
                }
            }
        }

        # Create the adventurer camp title
        create_landless_adventurer_title_tooltip_effect = yes

        # This event does the actual government change, title handoff, and full setup
        trigger_event = { id = ep3_laamps.0030 }
    }

    ai_check_interval = 0
    ai_will_do = 0
}
