	sort_order = 312

	selection_tooltip = pcm_b_add_building_slot_7_sub_realm_decision_tooltip
	confirm_text = pcm_b_add_building_slot_7_sub_realm_decision_ct
	ai_will_do = { base = 0 }
	ai_check_interval = 0

	is_shown = {
		is_landed = yes
		is_ai = no
		has_character_flag = pcm_b_show_decision_flag
	}

	# widget copied from create_holy_order_decision
	widget = {
		controller = create_holy_order
		title_valid = {
			trigger_if = {
				limit = { exists = this }
				abs_barony_is_valid_trigger = { CHARACTER = scope:ruler }
			}
			trigger_else = {
				custom_description = {
					text = "exists != this custom_decision"
					always = no
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		is_alive = yes
		is_imprisoned = no
		is_adult = yes
	}

	is_valid = {
	}

	cost = {
		gold = {
			value = 1
			if = {
				limit = {
					has_variable = abs_add_building_slot_cost_modifier
				}
				add = var:abs_add_building_slot_cost_modifier
			}
		}
	}

	effect = {
		scope:barony.title_province = {
			add_province_modifier = extra_building_slot
		}
		hidden_effect = {
			if = {
				limit = {
					NOT = { has_variable = abs_add_building_slot_cost_modifier }
				} 
				set_variable = {
					name = abs_add_building_slot_cost_modifier
					value = 1
				}
			} else = {
				change_variable = {
					name = abs_add_building_slot_cost_modifier
					add = 1
				}
			}
		}
	}
}
