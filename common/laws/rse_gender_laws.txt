# Realm Succession Events (RSE) gender law overrides.
# CK3 1.20: laws are flat entries with law_group_type/index; the group lives in common/law_groups (vanilla, unchanged).
# Rebuilt from vanilla 1.20 laws; the only Skonester change is the rse_flag_* bypass in can_keep.

male_only_law = {
	law_group_type = succession_gender_laws
	index = 0

	can_title_have = {
		can_title_have_law_general_trigger = yes
	}
	should_show_for_title = { # Never show this in the UI, just apply it through script
		always = no
	}
	can_keep = {
		# Skonester (RSE): characters flagged rse_flag_male_only keep this law regardless of faith/culture
		OR = {
			has_character_flag = rse_flag_male_only
			AND = {
				trigger_if = {
					limit = {
						can_have_male_monastic_holy_order_succession_law_trigger = yes
					}
					always = yes
				}
				trigger_else_if = {
					limit = {
						government_has_flag = government_is_theocracy
					}
					rite = { rite_has_parameter = clergy_must_be_male }
				}
				trigger_else_if = {
					limit = { should_synchronize_liege_law = yes }
					top_liege = { has_realm_law = male_only_law }
				}
				trigger_else = {
					custom_tooltip = {
						text = culture_not_male_only_tt
						rite = { rite_has_parameter = male_dominated_law }
						NOT = {
							culture = { has_cultural_parameter = female_only_inheritance }
						}
					}
				}
	
			}
		}
	}
	can_have = {
		trigger_if = {
			limit = {
				government_has_flag = government_is_theocracy
				NOT = { can_have_male_monastic_holy_order_succession_law_trigger = yes }
			}
			rite = { rite_has_parameter = clergy_must_be_male }
		}
	}
	can_pass = {
		can_change_succession_law_trigger = yes
	}
	should_start_with = {
		trigger_if = {
			limit = {
				can_have_male_monastic_holy_order_succession_law_trigger = yes
			}
			always = yes
		}
		trigger_else_if = {
			limit = { should_synchronize_liege_law = yes }
			top_liege = { has_realm_law = male_only_law }
		}
		trigger_else_if = {
			limit = {
				government_has_flag = government_has_merit
			}
			rite = {
				rite_has_parameter = male_dominated_law
			}
		}
		trigger_else_if = {
			limit = {
				government_has_flag = government_is_theocracy
			}
			rite = {
				rite_has_parameter = clergy_must_be_male
			}
		}
		trigger_else = {
			AND = {
				faith = {
					religion_tag = islam_religion
				}
				rite = {
					rite_has_parameter = male_dominated_law
				}
			}
		}
	}
	succession = {
		gender_law = male_only
	}
	flag = men_can_have_multiple_spouses
	flag = men_can_have_consorts
	flag = no_female_commanders

	pass_cost = {
		prestige = change_succession_law_prestige_cost
	}
	revoke_cost = {
		prestige = change_succession_law_prestige_cost
	}
}

male_preference_law = {
	law_group_type = succession_gender_laws
	index = 1

	can_title_have = {

	}
	should_show_for_title = { # Never show this in the UI, just apply it through script
		always = no
	}
	can_keep = {
		# Skonester (RSE): characters flagged rse_flag_male_preference keep this law regardless of faith/culture
		OR = {
			has_character_flag = rse_flag_male_preference
			AND = {
				trigger_if = {
					limit = {
						government_has_flag = government_is_theocracy
					}
					custom_tooltip = {
						text = gender_preference_law_not_for_clergy_tt

					}

							NOT = { rite_has_parameter = female_dominated_law }
						}
					}
					custom_tooltip = {
						text = male_preference_law_can_keep_faith_or_culture_trigger
						OR = {
							rite = { rite_has_parameter = male_dominated_law }
							NOT = {
								culture = { has_cultural_parameter = female_only_inheritance }
							}
						}
					}
				}
	
			}
		}
	}
	can_have = {
		trigger_if = {
			limit = {
				government_has_flag = government_is_theocracy
			}
			custom_tooltip = {
				text = gender_preference_law_not_for_clergy_tt
				always = no
			}
		}
	}
	can_pass = {
		can_change_succession_law_trigger = yes
	}
	should_start_with = {
		trigger_if = {
			limit = { should_synchronize_liege_law = yes }
			top_liege = { has_realm_law = male_preference_law }
		}
		trigger_else = {
			NOT = {
				culture = { has_cultural_parameter = female_preference_inheritance }
			}
			OR = {
				faith = {
					NOR = {
						religion_tag = islam_religion
						AND = {
							root = {
								government_has_flag = government_has_merit
							}
							root.rite = {
								rite_has_parameter = male_dominated_law
							}
						}
					}
					root.rite = {
						rite_has_parameter = male_dominated_law
					}
				}
				AND = {
					has_realm_law = male_only_law # For when male_only_law ceases to be a legal option
					trigger_if = {
						limit = { should_synchronize_liege_law = yes }
						NOT = {
							top_liege = { has_realm_law = male_only_law }
						}
					}
					trigger_else = {
						custom_tooltip = {
							text = culture_not_female_only_tt
							NOR = {
								rite = { rite_has_parameter = male_dominated_law }
								NOT = {
									culture = { has_cultural_parameter = female_only_inheritance }
								}
							}
						}
					}
				}
			}
		}
	}
	succession = {
		gender_law = male_preference
	}
	flag = men_can_have_multiple_spouses
	flag = men_can_have_consorts
	flag = no_female_commanders

	pass_cost = {
		prestige = change_succession_law_prestige_cost
	}
	revoke_cost = {
		prestige = change_succession_law_prestige_cost
	}
}

equal_law = {
	law_group_type = succession_gender_laws
	index = 2

	can_title_have = {
		can_title_have_law_general_trigger = yes
	}
	should_show_for_title = { # Never show this in the UI, just apply it through script
		always = no
	}
	can_keep = {
		# Skonester (RSE): characters flagged rse_flag_equal keep this law regardless of faith/culture
		OR = {
			has_character_flag = rse_flag_equal
			AND = {
				trigger_if = {
					limit = {
						government_has_flag = government_is_theocracy
					}
					rite = { rite_has_parameter = clergy_can_be_either_gender }
				}
				trigger_else_if = {
					limit = { should_synchronize_liege_law = yes }
					top_liege = { has_realm_law = equal_law }
				}
				trigger_else = {
					custom_tooltip = {
						text = equal_law_culture_faith_or_innovation_trigger
						OR = {
							culture = { has_cultural_parameter = gender_equal_inheritance }
							#culture = { has_innovation = innovation_visigothic_codes }
							rite = { rite_has_parameter = gender_equal_law }
							AND = { #Culture is female dominated but religion is male dominated
								rite = { rite_has_parameter = male_dominated_law }
								culture = { has_cultural_parameter = female_only_inheritance }
							}
							AND = { #Culture is male dominated but faith is female dominated
								rite = { rite_has_parameter = female_dominated_law }
								NOT = { culture = { has_cultural_parameter = female_only_inheritance } }
							}
							primary_title.var:can_keep_realm_law ?= flag:equal_law
						}
					}
				}
	
			}
		}
	}
	can_have = {
		trigger_if = {
			limit = {
				government_has_flag = government_is_theocracy
			}
			rite = { rite_has_parameter = clergy_can_be_either_gender }
		}
	}
	can_pass = {
		#can_change_to_equal_succession_law_trigger = yes
		can_change_succession_law_trigger = yes
	}
	should_start_with = {
		trigger_if = {
			limit = { should_synchronize_liege_law = yes }
			top_liege = { has_realm_law = equal_law }
		}
		trigger_else_if = {
			limit = {
				government_has_flag = government_is_theocracy
			}
			rite = { rite_has_parameter = clergy_can_be_either_gender }
		}
		trigger_else = {
			NOR = {
				culture = { has_cultural_parameter = female_preference_inheritance }
				culture = { has_cultural_tradition = tradition_nubian_warrior_queens }
			}
			rite = {
				rite_has_parameter = gender_equal_law
			}
			NOT = { has_realm_law = female_only_law } # For when female_only_law ceases to be valid
		}
	}
	succession = {
		gender_law = equal
	}
	flag = men_can_have_multiple_spouses
	flag = men_can_have_consorts
	flag = women_can_have_multiple_spouses
	flag = women_can_have_consorts

	pass_cost = {
		prestige = change_succession_law_prestige_cost
	}
	revoke_cost = {
		prestige = change_succession_law_prestige_cost
	}

	ai_will_do = {
		if = {
			limit = {
				exists = primary_title
				primary_title.tier > tier_county
				rite = {
					NOT = {
						rite_has_dominant_ruling_gender = prev
					}
				}
			}
			value = 2
		}
	}
}
