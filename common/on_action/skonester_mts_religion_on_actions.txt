# More Tenets Slots：创建信仰时清除占位空槽
# CK3 1.20: on_faith_created now runs with ROOT = the new faith (in 1.19 ROOT was the creating character),
# so the cleanup runs as an effect in faith scope instead of firing a character event.
# (skonester_mts_faith_creation.001 is kept for reference but no longer hooked.)

  on_actions = { skonester_on_mts_faith_creation }

# CK3 1.20: players can also create a new Rite inside an existing faith, which fires on_rite_created
# (ROOT = the creating character, `rite` = the new rite) instead of on_faith_created. Clear the placeholder
# tenets from that rite too. remove_rite_tenet is the rite-scope version of remove_tenet.
on_rite_created = {
  on_actions = { skonester_on_mts_rite_creation }
}

skonester_on_mts_rite_creation = {
  effect = {
    rite ?= {
      if = { limit = { rite_has_tenet = tenet_zz_empty_1 } remove_rite_tenet = tenet:tenet_zz_empty_1 }
