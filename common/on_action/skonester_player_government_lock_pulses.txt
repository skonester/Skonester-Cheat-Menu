# 人类玩家政体锁定：开局记录；日/月脉冲见 zzz_skonester_fidelity_affair_sweep_daily 与 skonester_realm_authority_lock_pulses

on_game_start_after_lobby = {
  on_actions = {
    skonester_player_government_lock_startup_pulse
  }
}

skonester_player_government_lock_startup_pulse = {
  effect = {
    every_player = {
      skonester_player_government_lock_apply_effect = yes
    }
  }
}

skonester_player_government_lock_monthly_pulse = {
  effect = {
    root = {
      if = {
        limit = {
          is_ai = no
          NOT = { has_character_flag = skonester_player_government_locked }
          skonester_player_government_lock_eligible_trigger = yes
        }
        skonester_player_government_lock_apply_effect = yes
      }
      skonester_player_government_lock_maintain_effect = yes
    }
  }
}

skonester_player_government_lock_daily_pulse = {
  effect = {
    root = {
      if = {
        limit = {
          is_ai = no
          NOT = { has_character_flag = skonester_player_government_locked }
          skonester_player_government_lock_eligible_trigger = yes
        }
        skonester_player_government_lock_apply_effect = yes
      }
      skonester_player_government_lock_maintain_effect = yes
    }
  }
}
