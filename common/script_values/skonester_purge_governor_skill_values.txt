  value = 0
  add = diplomacy
  add = stewardship
  add = learning
}

skonester_military_skill_sum_for_purge_value = {
  value = 0
  add = martial
  add = prowess
}

skonester_civil_skill_sum_for_purge_effective_value = {
  value = skonester_civil_skill_sum_for_purge_value
  if = {
    limit = { government_has_flag = sansizhi_gov }
    subtract = 5
  }
}

  value = skonester_military_skill_sum_for_purge_value
  if = {
    limit = { government_has_flag = weisuozhi_gov }
    subtract = 5
  }
}

# 读取 scope:skonester_grant_target_title 档位；无 scope 时按男爵线
skonester_civil_grant_threshold_for_target_title_value = {
  if = {
    limit = {
      exists = scope:skonester_grant_target_title
      scope:skonester_grant_target_title = {
        tier = tier_empire
      }
    }
    value = skonester_civil_grant_empire_threshold_value
  }
  else_if = {
    limit = {
      exists = scope:skonester_grant_target_title
      scope:skonester_grant_target_title = {
        tier = tier_kingdom
      }
    }
    value = skonester_civil_grant_kingdom_threshold_value
  }
  else_if = {
    limit = {
      exists = scope:skonester_grant_target_title
      scope:skonester_grant_target_title = {
        tier = tier_duchy
      }
    }
    value = skonester_civil_grant_duchy_threshold_value
  }
  else_if = {
    limit = {
      exists = scope:skonester_grant_target_title
      scope:skonester_grant_target_title = {
        tier = tier_county
      }
    }
    value = skonester_civil_grant_county_threshold_value
  }
  else = {
    value = skonester_civil_grant_barony_threshold_value
  }
}
