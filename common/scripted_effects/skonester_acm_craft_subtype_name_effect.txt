# 子类默认名：制作旗标在 skonester_acm_craft_creator（或 scope:actor）上
skonester_acm_craft_subtype_has_flag_effect = {
  if = {
    limit = {
      OR = {
        AND = {
          exists = scope:skonester_acm_craft_creator
          scope:skonester_acm_craft_creator = { has_character_flag = $FLAG$ }
        }
        AND = {
          exists = scope:actor
          scope:actor = { has_character_flag = $FLAG$ }
        }
      }
    }
    set_artifact_name = $NAME$
    set_artifact_description = skonester_acm_ades_craft_subtype
  }
}
