}

# 核心信条：记录被点击槽位，填充全部核心信条供替换
skonester_acm_faith_begin_pick_core_tenets_effect = {
  skonester_acm_faith_init_group_catalog_effect = yes
  remove_character_flag = skonester_acm_faith_doctrine_picker_open
  add_character_flag = skonester_acm_faith_pick_sidebar_open
  skonester_acm_faith_capture_pick_doctrine_effect = yes
  if = {
    limit = { has_variable = skonester_acm_pick_doctrine_key }
    set_variable = {
      name = skonester_acm_pick_slot
      value = var:skonester_acm_pick_doctrine_key
    }
  }
  else = {
    remove_variable = skonester_acm_pick_slot
  }
  set_variable = {
    name = skonester_acm_pick_group_name
    value = flag:doctrine_core_tenets
  }
  skonester_acm_faith_populate_pick_current_effect = yes
}

skonester_acm_faith_apply_pick_doctrine_effect = {
  skonester_acm_faith_capture_pick_doctrine_effect = yes
  # 始终以「新选教义」解析组名，避免 begin_pick 时旧组名导致 remove 未命中
  remove_variable = skonester_acm_pick_group_name
  skonester_acm_faith_resolve_pick_group_effect = yes
  skonester_acm_faith_resolve_pick_group_from_key_fallback_effect = yes
  if = {
    limit = {
      has_variable = skonester_acm_pick_group_name
      var:skonester_acm_pick_group_name = flag:doctrine_core_tenets
    }
    skonester_acm_faith_remove_pick_slot_doctrine_effect = yes
  }
  else_if = {
    limit = { has_variable = skonester_acm_pick_group_name }
    skonester_acm_faith_remove_pick_group_members_effect = yes
  }
  skonester_acm_faith_add_doctrine_to_selected_effect = yes
  remove_variable = skonester_acm_pick_slot
  remove_variable = skonester_acm_pick_group_name
  remove_variable = skonester_acm_pick_doctrine_key
  skonester_acm_faith_refresh_display_lists_effect = yes
}

skonester_close_acm_faith_doctrine_picker_effect = {

  skonester_acm_faith_clear_pick_category_flags_effect = yes
  remove_character_flag = skonester_acm_faith_pick_sidebar_open
}
