skonester_acm_cycle_craft_m1_effect = {
  if = {
    limit = { NOT = { has_variable = skonester_craft_m1 } }
    set_variable = { name = skonester_craft_m1 value = 0 }
  }
  change_variable = { name = skonester_craft_m1 add = 1 }
  if = {
    limit = { var:skonester_craft_m1 > 64 }
    set_variable = { name = skonester_craft_m1 value = 0 }
  }
  if = {
    limit = { var:skonester_craft_m1 = 0 }
    set_variable = { name = skonester_craft_m1_cnt value = 0 }
  }
  if = {
    limit = { NOT = { var:skonester_craft_m1 = 0 } }
    if = {
      limit = {
        OR = {
          NOT = { has_variable = skonester_craft_m1_cnt }
          var:skonester_craft_m1_cnt = 0
        }
      }
      set_variable = { name = skonester_craft_m1_cnt value = 1 }
    }
  }
}

skonester_acm_inc_craft_m1_cnt_effect = {
  if = {
    limit = { NOT = { var:skonester_craft_m1 = 0 } }
    skonester_acm_sync_craft_layer_quality_effect = yes
    if = {
      limit = { NOT = { has_variable = skonester_craft_m1_cnt } }
      set_variable = {
        name = skonester_craft_m1_cnt
        value = var:skonester_craft_quality_min_layers
      }
    }
    if = {
      limit = {
        var:skonester_craft_m1_cnt < var:skonester_craft_quality_min_layers
      }
      set_variable = {
        name = skonester_craft_m1_cnt
        value = var:skonester_craft_quality_min_layers
      }
    }
    change_variable = { name = skonester_craft_m1_cnt add = 1 }
    if = {
      limit = { var:skonester_craft_m1_cnt > 999 }
      set_variable = { name = skonester_craft_m1_cnt value = 999 }
    }
  }
}

    if = {
      limit = { var:skonester_craft_m5_cnt < var:skonester_craft_quality_min_layers }
      set_variable = {
        name = skonester_craft_m5_cnt
        value = var:skonester_craft_quality_min_layers
      }
    }
  }
}
