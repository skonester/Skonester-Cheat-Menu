# 宝物修改：列出目标宝物 → 修改 GUI（名称/描述/品质/修饰）→ 保存
# 变量 skonester_acm_repair_victim 必须写在 scope:actor（玩家）上，事件用 var:skonester_acm_repair_victim 读取（同窃宝链）。

# 将所选宝物名称/描述载入玩家编辑缓冲（与制作窗同一套 key；优先 creator 再 artifact）
skonester_acm_sync_edit_text_from_artifact_effect = {
  remove_character_flag = skonester_acm_edit_name_confirmed
  remove_character_flag = skonester_acm_edit_desc_confirmed
  if = {
    limit = { exists = global_var:skonester_acm_repair_target }
    global_var:skonester_acm_repair_target = {
      if = {
        limit = {
          exists = creator
          creator = { has_character_flag = skonester_acm_edit_name_confirmed }
        }
        creator = { save_scope_as = skonester_acm_name_copy_source }
      }
      else_if = {
        limit = { has_variable = skonester_acm_has_custom_name }
        save_scope_as = skonester_acm_name_copy_source
      }
      if = {
        limit = {
          exists = creator
          creator = { has_character_flag = skonester_acm_edit_desc_confirmed }
        }
        creator = { save_scope_as = skonester_acm_desc_copy_source }
      }
      else_if = {
        limit = { has_variable = skonester_acm_has_custom_desc }
        save_scope_as = skonester_acm_desc_copy_source
      }
    }
  }
  root = {
    remove_localized_text = skonester_acm_edit_artifact_name
    remove_localized_text = skonester_acm_edit_artifact_desc
    if = {
      limit = { exists = scope:skonester_acm_name_copy_source }
      copy_localized_text = {
        key = skonester_acm_edit_artifact_name
        target = scope:skonester_acm_name_copy_source
      }
    }
    if = {
      limit = { exists = scope:skonester_acm_desc_copy_source }
      copy_localized_text = {
        key = skonester_acm_edit_artifact_desc
        target = scope:skonester_acm_desc_copy_source
      }
    }
    if = {
      limit = { exists = scope:skonester_acm_name_copy_source }
      add_character_flag = skonester_acm_edit_name_confirmed
    }
    if = {
      limit = { exists = scope:skonester_acm_desc_copy_source }
      add_character_flag = skonester_acm_edit_desc_confirmed
    }
    skonester_acm_refresh_artifact_text_preview_effect = yes
  }
}

skonester_acm_register_repair_target_effect = {
  save_scope_as = skonester_acm_picked_artifact
  set_global_variable = {
    name = skonester_acm_repair_target
    value = scope:skonester_acm_picked_artifact

  root = {

      name = skonester_acm_repair_artifact_var

    }
  }
}

# 将所选宝物当前品质同步到 GUI 品质按钮（修饰槽仍由玩家重新选择）
skonester_acm_sync_repair_gui_from_artifact_effect = {
  if = {
    limit = { exists = global_var:skonester_acm_repair_target }
    global_var:skonester_acm_repair_target = {
      if = {
        limit = { rarity = illustrious }
        root = { skonester_acm_set_quality_illustrious_effect = yes }
      }
      else_if = {
        limit = { rarity = famed }
        root = { skonester_acm_set_quality_famed_effect = yes }
      }
      else_if = {
        limit = { rarity = masterwork }
        root = { skonester_acm_set_quality_masterwork_effect = yes }
      }
      else = {
        root = { skonester_acm_set_quality_common_effect = yes }
      }
    }
  }
  else = {
    skonester_acm_set_quality_common_effect = yes
  }
}

skonester_acm_after_repair_pick_effect = {
  skonester_acm_repair_picker_close_effect = yes
  remove_character_flag = skonester_acm_modify_session
  remove_character_flag = skonester_acm_gui_open
  skonester_acm_sync_repair_gui_from_artifact_effect = yes
  skonester_acm_sync_edit_text_from_artifact_effect = yes
  skonester_acm_load_modifier_slots_from_artifact_effect = yes
  skonester_acm_save_repair_modifier_baseline_effect = yes

    limit = {
      exists = global_var:skonester_acm_repair_target
      global_var:skonester_acm_repair_target = { has_variable = skonester_acm_from_gui_craft }
    }
    root = { save_scope_as = skonester_acm_craft_creator }
    global_var:skonester_acm_repair_target = {
      save_scope_as = skonester_new_crafted_artifact
      skonester_acm_persist_craft_slots_on_artifact_effect = yes
    }
  }
  add_character_flag = skonester_acm_repair_gui_open
  if = {
    limit = { exists = global_var:skonester_acm_repair_target }
    send_interface_toast = {
      type = event_toast_effect_neutral
      title = skonester_acm_repair_selected_toast_t
      left_icon = root
      right_icon = global_var:skonester_acm_repair_target
    }
  }
}

# 在玩家（或 scope:actor 上下文）上打开宝物列表；勿包 scope:actor，否则 GUI 按钮点击时 actor 不存在导致无反应
skonester_acm_gui_open_repair_picker_effect = {
  if = {
    limit = { exists = global_var:skonester_cheat_menu_recipient }
    set_variable = {
      name = skonester_acm_repair_victim
      value = global_var:skonester_cheat_menu_recipient
    }
  }
  else_if = {
    limit = { exists = global_var:skonester_acm_recipient }
    set_variable = {
      name = skonester_acm_repair_victim
      value = global_var:skonester_acm_recipient
    }
  }
  else_if = {
    limit = { exists = scope:skonester_cheat_menu_target }
    set_variable = {
      name = skonester_acm_repair_victim
      value = scope:skonester_cheat_menu_target
    }
  }
  else_if = {
    limit = { exists = scope:recipient }
    set_variable = {
      name = skonester_acm_repair_victim
      value = scope:recipient
    }
  }
  set_variable = {
    name = skonester_acm_repair_page
    value = 0
  }
  if = {
    limit = { has_variable = skonester_acm_repair_art_total }
    remove_variable = skonester_acm_repair_art_total
  }
  add_character_flag = skonester_acm_repair_picker_active
  if = {
    limit = { has_global_variable = skonester_acm_repair_target }
    remove_global_variable = skonester_acm_repair_target
  }
  if = {
    limit = { has_variable = skonester_acm_repair_artifact_var }
    remove_variable = skonester_acm_repair_artifact_var
  }
  remove_character_flag = skonester_acm_repair_gui_open
  remove_character_flag = skonester_acm_gui_open
  skonester_acm_repair_picker_reopen_effect = yes
}

      save_scope_as = actor

        scope:skonester_acm_craft_creator = {
          has_variable = skonester_acm_repair_baseline_m1

      global_var:skonester_acm_repair_target = {
        save_scope_as = skonester_new_crafted_artifact
        skonester_acm_apply_craft_quality_rarity_from_craft_creator_effect = yes
        skonester_apply_crafted_artifact_modifiers_effect = yes

      }
      if = {
        limit = { has_variable = skonester_acm_repair_baseline_m3 }
        remove_variable = skonester_acm_repair_baseline_m3
      }
      if = {
        limit = { has_variable = skonester_acm_repair_baseline_m4 }
        remove_variable = skonester_acm_repair_baseline_m4
      }
      if = {
        limit = { has_variable = skonester_acm_repair_baseline_m5 }
        remove_variable = skonester_acm_repair_baseline_m5
      }
      if = {
        limit = { has_variable = skonester_acm_repair_baseline_m1_cnt }
        remove_variable = skonester_acm_repair_baseline_m1_cnt
