# Skonester右键作弊菜单：统一用 global_var 锁定右击对象（与 skonester_acm_recipient 同套路）。
# interaction 的 save_scope_as 不会随 trigger_event 传入事件，导致「点了没反应」。

skonester_cheat_menu_lock_recipient_effect = {
  if = {
    limit = { exists = scope:recipient }
    set_global_variable = {
      name = skonester_cheat_menu_recipient
      value = scope:recipient
    }
  }
  else_if = {
    limit = { exists = scope:skonester_cheat_menu_target }
    set_global_variable = {
      name = skonester_cheat_menu_recipient
      value = scope:skonester_cheat_menu_target
    }
  }
  else_if = {
    limit = { exists = scope:actor }
    set_global_variable = {
      name = skonester_cheat_menu_recipient
      value = scope:actor
    }
  }
  else_if = {
    limit = { has_global_variable = skonester_cheat_menu_recipient }
  }
}

skonester_cheat_menu_sync_target_effect = {
  if = {
    limit = { exists = global_var:skonester_cheat_menu_recipient }
    global_var:skonester_cheat_menu_recipient = {
      save_scope_as = skonester_cheat_menu_target
    }
  }
  else_if = {
    limit = { exists = scope:recipient }
    scope:recipient = {
      save_scope_as = skonester_cheat_menu_target
    }
  }
  else_if = {
    limit = {
      exists = root
      root = { has_variable = skonester_military_menu_victim }
    }
    root = {
      var:skonester_military_menu_victim = { save_scope_as = skonester_cheat_menu_target }
    }
  }
  else_if = {
    limit = { exists = scope:actor }
    scope:actor = {
      save_scope_as = skonester_cheat_menu_target
    }
  }
}

skonester_cheat_menu_clear_recipient_effect = {
  if = {
    limit = { has_global_variable = skonester_cheat_menu_recipient }
    remove_global_variable = skonester_cheat_menu_recipient
  }
  if = {
    limit = {
      exists = root
      root = { has_variable = skonester_military_menu_victim }
    }
    root = { remove_variable = skonester_military_menu_victim }
  }
}

          }
        }

    }
  }
}

skonester_cheat_menu_open_artifact_craft_effect = {
  skonester_cheat_menu_lock_recipient_effect = yes
  if = {
    limit = {
      exists = scope:actor
      scope:actor = { skonester_player_may_use_artifact_craft_trigger = yes }
    }
  scope:actor = {
    save_scope_as = skonester_acm_craft_creator
    remove_character_flag = skonester_acm_modify_session
    skonester_cheat_menu_sync_target_effect = yes
    if = {
      limit = { exists = scope:skonester_cheat_menu_target }
      set_global_variable = {
        name = skonester_acm_recipient
        value = scope:skonester_cheat_menu_target
      }
    }
    else_if = {
      limit = { exists = scope:recipient }
      set_global_variable = {
        name = skonester_acm_recipient
        value = scope:recipient
      }
    }
    add_character_flag = skonester_craft_give_to_recipient
    skonester_craft_init_modifier_slot_defaults_effect = yes
    if = {
      limit = { NOT = { has_variable = skonester_acm_artifact_type } }
      skonester_acm_init_artifact_type_defaults_effect = yes
    }
    if = {
      limit = { NOT = { has_character_flag = skonester_artifact_quality_set } }
      skonester_acm_set_quality_common_effect = yes
    }
    add_character_flag = skonester_acm_gui_open
  }
  }
}

# Skonester菜单 0030：打开 advcm 式宝物制作窗（与右键「宝物制作」共用变量/修饰列表）
skonester_cheat_menu_open_artifact_craft_gui_effect = {
  if = {
    limit = {
      exists = scope:actor
      scope:actor = { skonester_player_may_use_artifact_craft_trigger = yes }
    }
  scope:actor = {
    save_scope_as = skonester_acm_craft_creator
    skonester_craft_init_modifier_slot_defaults_effect = yes
    if = {
      limit = { NOT = { has_variable = skonester_acm_artifact_type } }
      skonester_acm_init_artifact_type_defaults_effect = yes
    }
    if = {
      limit = { NOT = { has_character_flag = skonester_artifact_quality_set } }
      skonester_acm_set_quality_common_effect = yes
    }
    if = {
      limit = {
        has_character_flag = skonester_craft_give_to_recipient
        exists = scope:recipient
        NOT = { has_global_variable = skonester_acm_recipient }
      }
      set_global_variable = {
        name = skonester_acm_recipient
        value = scope:recipient
      }
    }
    add_character_flag = skonester_acm_gui_open
  }
  }
}
