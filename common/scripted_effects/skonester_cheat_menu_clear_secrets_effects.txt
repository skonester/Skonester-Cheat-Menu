# Skonester「清除秘密」：在 root 为待清理角色时，移除其持有/被指向/已知的全部秘密。
# 使用 while + random_*_secret 逐条删除，避免在 every_* 迭代中就地 remove 导致漏删或引擎行为异常。
skonester_cheat_menu_clear_all_secrets_on_root_effect = {
  while = {
    count = 500
    limit = {
      any_secret = {
        always = yes
      }
    }
    random_secret = {
      remove_secret = yes
    }
  }
  while = {
    count = 500
    limit = {
      any_targeting_secret = {
        always = yes
      }
    }
    random_targeting_secret = {
      remove_secret = yes
    }
  }
  while = {
    count = 500
    limit = {
      any_known_secret = {
        always = yes
      }
    }
    random_known_secret = {
      remove_secret = yes
    }
  }
}
