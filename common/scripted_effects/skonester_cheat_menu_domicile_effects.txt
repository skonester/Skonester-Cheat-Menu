# Skonester「庄园编辑」：须在 scope:recipient 或 scope:skonester_cheat_menu_target 上调用。
# 外槽/内建沿用原版 fill_external_* 与 add_random_internal_*；主建筑逐级 add_domicile_building。

# 行政/贤能等：庄园多在世族头衔或宗族族长名下；帝国/公国/伯爵持有者本人常无 domicile。
skonester_cheat_menu_resolve_domicile_holder_effect = {
  if = {
    limit = {
      is_alive = yes
      has_domicile = yes
    }
    save_scope_as = skonester_domicile_holder
  }
  else_if = {
    limit = {
      is_alive = yes
      exists = house.house_head
      house.house_head = {
        is_alive = yes
        has_domicile = yes
      }
    }
    house.house_head = {
      save_scope_as = skonester_domicile_holder
    }
  }
  else_if = {
    limit = {
      is_alive = yes
      exists = house
    }
    house = {
      ordered_house_member = {
        limit = {
          is_alive = yes
          has_domicile = yes
          is_house_head = yes
        }
        order_by = age
        save_scope_as = skonester_domicile_holder
      }
    }
  }
  else_if = {
    limit = {
      is_alive = yes
      exists = house
    }
    house = {
      ordered_house_member = {
        limit = {
          is_alive = yes
          has_domicile = yes
          any_held_title = { is_noble_family_title = yes }
        }
        order_by = age
        save_scope_as = skonester_domicile_holder
      }
    }
  }
  else_if = {
    limit = {
      is_alive = yes
      exists = house
    }
    house = {
      random_house_member = {
        limit = {
          is_alive = yes
          has_domicile = yes
        }
        save_scope_as = skonester_domicile_holder
      }
    }
  }
}

skonester_cheat_menu_domicile_warn_no_holder_effect = {
  if = {
    limit = { NOT = { exists = scope:skonester_domicile_holder } }
    root = {
      send_interface_toast = {
        type = event_toast_effect_bad
        title = skonester_cheat_menu_fill_domicile_must_have_estate_tt
        left_icon = root
      }
    }
  }
}

# 绕过 can_construct 革新门槛与建造费用（行政制国库 / 西式金币）。
skonester_domicile_cheat_prereqs_effect = {
  owner = {
    if = {
      limit = { has_treasury = yes }
      add_treasury = 50000
    }
    else = {
      add_gold = 50000
    }
    culture = {
      if = {
        limit = { NOT = { has_innovation = innovation_city_planning } }
        add_innovation = innovation_city_planning
      }
      if = {
        limit = { NOT = { has_innovation = innovation_manorialism } }
        add_innovation = innovation_manorialism
      }
      if = {
        limit = { NOT = { has_innovation = innovation_development_03 } }
        add_innovation = innovation_development_03
      }
      if = {
        limit = { NOT = { has_innovation = innovation_cranes } }
        add_innovation = innovation_cranes
      }
    }
  }
  switch = {
    trigger = is_domicile_type
    estate = {
      if = {
        limit = { NOT = { has_domicile_building_or_higher = estate_main_01 } }
        add_domicile_building = estate_main_01
      }
    }
    east_asian_estate = {
      if = {
        limit = { NOT = { has_domicile_building_or_higher = east_asian_estate_main_01 } }
        add_domicile_building = east_asian_estate_main_01
      }
    }
    japanese_manor = {
      if = {
        limit = { NOT = { has_domicile_building_or_higher = japanese_manor_main_01 } }
        add_domicile_building = japanese_manor_main_01
      }
    }
  }
}

  scope:recipient = {
    if = {
      limit = {
        is_alive = yes

        # 任命宰相 (Chancellor)
        if = {
          limit = { NOT = { has_council_position = councillor_chancellor } }

            limit = {
              is_adult = yes
              is_imprisoned = no
              NOT = { is_incapable = yes }
              NOT = { this = scope:recipient }
              NOT = { is_councillor_of = scope:recipient }
            }
            add_character_modifier = {
              modifier = skonester_court_cabinet_chancellor_candidate
              days = 1
            }
          }
          ordered_vassal = {
            limit = {
              has_character_modifier = skonester_court_cabinet_chancellor_candidate
            }
            order_by = aptitude:councillor_chancellor
            max = 1
            check_range_bounds = no
            save_scope_as = skonester_pos_pick
          }
          if = {
            limit = { exists = scope:skonester_pos_pick }
            appoint_councillor = scope:skonester_pos_pick
          }
          every_vassal = {
