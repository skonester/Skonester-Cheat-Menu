skonester_cheat_menu_add_dynasty_legacy_perk_if_missing_effect = {
  if = {
    limit = { NOT = { has_dynasty_perk = $PERK$ } }
    add_dynasty_perk = $PERK$
  }
}

skonester_cheat_menu_clear_dynasty_renown_after_legacies_effect = {
  add_dynasty_prestige = {
    value = 0
    subtract = dynasty_prestige
  }
  while = {
    limit = { dynasty_prestige_level > 0 }
    add_dynasty_prestige_level = -1
  }
}
