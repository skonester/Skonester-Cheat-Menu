# 亲密 / 强制怀孕共用：必定进入怀孕状态
skonester_guaranteed_pregnancy_effect = {
  if = {
    limit = {
      is_female = yes
      is_adult = yes
    }
    if = {
      limit = { is_pregnant = yes }
      end_pregnancy = yes
    }
    if = {
      limit = {
        skonester_stealth_cheat_pregnancy_trigger = yes
      }
      make_pregnant = {
        father = scope:actor
        known_bastard = no
      }
      set_variable = {
        name = skonester_stealth_fp_bio_dad
        value = scope:actor
      }
      skonester_force_pregnancy_set_stealth_assumed_father_effect = yes
    }
    else = {
      make_pregnant = {
        father = scope:actor
      }
      set_pregnancy_assumed_father = scope:actor
    }
    if = {
      limit = { is_pregnant = no }
      make_pregnant = {
        father = scope:actor
      }
      if = {
        limit = {
          skonester_stealth_cheat_pregnancy_trigger = yes
        }
        set_variable = {
          name = skonester_stealth_fp_bio_dad
          value = scope:actor
        }
        skonester_force_pregnancy_set_stealth_assumed_father_effect = yes
      }
      else = {
        set_pregnancy_assumed_father = scope:actor
      }
    }
  }
}

    }
  }
  else_if = {
    limit = { pregnancy_real_father = { always = yes } }

  }
  else_if = {
    limit = { pregnancy_assumed_father = { always = yes } }
