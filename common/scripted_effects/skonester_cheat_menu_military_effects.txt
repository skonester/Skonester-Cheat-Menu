# 同步右击目标；优先 global_var，其次玩家身上的 skonester_military_menu_victim
skonester_military_ensure_target_effect = {
  skonester_cheat_menu_sync_target_effect = yes
  if = {
    limit = { NOT = { exists = scope:skonester_cheat_menu_target } }
    root = {
      if = {
        limit = { has_variable = skonester_military_menu_victim }
        var:skonester_military_menu_victim = {
          save_scope_as = skonester_cheat_menu_target
          set_global_variable = {
            name = skonester_cheat_menu_recipient
            value = this
          }
        }
      }
    }
  }
}

skonester_military_send_supply_toggle_toast_effect = {
  if = {
    limit = {
      exists = scope:skonester_military_toggle_char
      scope:skonester_military_toggle_char = { var:skonester_military_toggle_on = 1 }
    }
    root = {
      send_interface_toast = {
        type = event_toast_effect_good
        title = skonester_military_supply_on_toast
        desc = skonester_military_supply_on_toast_desc
        left_icon = scope:skonester_military_toggle_char
      }
    }
  }
  else_if = {
    limit = { exists = scope:skonester_military_toggle_char }
    root = {
      send_interface_toast = {
        type = event_toast_effect_neutral
        title = skonester_military_supply_off_toast
        left_icon = scope:skonester_military_toggle_char
      }
    }
  }
}

skonester_cheat_menu_raise_all_levies_effect = {
  skonester_military_ensure_target_effect = yes
  if = {

    scope:skonester_cheat_menu_target = {
      skonester_cheat_menu_raise_all_levies_on_character_effect = yes
      save_scope_as = skonester_military_toast_char
    }
    root = {
      send_interface_toast = {
        type = event_toast_effect_good
        title = skonester_military_raise_levies_toast
        desc = skonester_military_raise_levies_toast_desc
        left_icon = scope:skonester_military_toast_char
      }
    }
  }
}

skonester_cheat_menu_raise_all_levies_on_character_effect = {
  if = {
    limit = {
      is_alive = yes
      exists = capital_province
    }
    capital_province = { save_scope_as = skonester_military_spawn_loc }
  }
  else_if = {
    limit = {
      is_alive = yes
      exists = location
    }
    location = { save_scope_as = skonester_military_spawn_loc }
  }
  if = {
    limit = { exists = scope:skonester_military_spawn_loc }
    spawn_army = {
      name = skonester_military_raise_levies_army_name
      levies = {
        add = max_military_strength
        min = 1000
      }
      location = scope:skonester_military_spawn_loc
      inheritable = no
    }
  }
}

skonester_cheat_menu_instant_army_replenish_on_character_effect = {
  # 不修改士兵/常备人数；切换军需修正 + stacking，并拉满当前军团补给
  if = {
    limit = { NOT = { has_character_modifier = skonester_military_instant_replenish_modifier } }
    skonester_cheat_menu_refill_all_army_supply_on_character_effect = yes
    skonester_cheat_menu_apply_supply_overflow_stacks_effect = yes
    add_character_modifier = skonester_military_instant_replenish_modifier
    skonester_cheat_menu_start_supply_war_loop_effect = yes
    skonester_military_cheat_authorize_target_if_player_actor_effect = yes
    save_scope_as = skonester_military_toggle_char
    set_variable = { name = skonester_military_toggle_on value = 1 }
  }
  else = {
    remove_character_modifier = skonester_military_instant_replenish_modifier
    remove_all_character_modifier_instances = skonester_military_supply_overflow_modifier
    save_scope_as = skonester_military_toggle_char
    set_variable = { name = skonester_military_toggle_on value = 0 }
  }
}

skonester_cheat_menu_fill_knights_on_character_effect = {
  save_scope_as = skonester_knight_fill_owner
  if = {
    limit = { number_of_knights < max_number_of_knights }
    every_courtier_or_guest = {
      limit = {
        is_knight = no
        prowess >= 8
        can_be_knight_trigger = { ARMY_OWNER = scope:skonester_knight_fill_owner }
      }
      set_knight_status = force
    }
  }
}
