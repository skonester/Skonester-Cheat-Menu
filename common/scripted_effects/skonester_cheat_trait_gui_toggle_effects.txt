skonester_cheat_trait_gui_toggle_immortal_effect = {
  scope:skonester_trait_edit_target = {
    if = {
      limit = { has_trait = skonester_trait_immortal_ageless }
      remove_trait = skonester_trait_immortal_ageless
    }
    else = {
      set_age = 25
      if = {
        limit = { has_trait = incapable }
        remove_trait = incapable
      }
      skonester_cheat_menu_grant_cheat_trait_effect = { TRAIT = skonester_trait_immortal_ageless }
    }
  }
}

skonester_cheat_trait_gui_toggle_yannian_dan_effect = {
  scope:skonester_trait_edit_target = {
    if = {
      limit = { has_character_flag = skonester_yannian_dan_active }
      remove_character_flag = skonester_yannian_dan_active
      if = {
        limit = { has_variable = skonester_yannian_dan_original_age }
        set_age = var:skonester_yannian_dan_original_age
      }
      if = {
        limit = { has_variable = skonester_yannian_dan_original_age }
        remove_variable = skonester_yannian_dan_original_age
      }
      if = {
        limit = { has_trait = skonester_trait_yannian_dan }
        remove_trait = skonester_trait_yannian_dan
      }
    }
    else = {
      add_character_flag = skonester_yannian_dan_active
      if = {
        limit = { NOT = { has_variable = skonester_yannian_dan_original_age } }
        set_variable = {
          name = skonester_yannian_dan_original_age
          value = age
        }
      }
      set_age = 25
      if = {
        limit = { has_trait = incapable }
        remove_trait = incapable
      }
      add_trait = skonester_trait_yannian_dan
    }
  }
  skonester_traits_authorize_recipient_if_player_actor_effect = yes
}
