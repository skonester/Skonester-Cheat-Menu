# 伯爵领作弊：按 actor 上的 domain/vassal 旗标对 skonester_cnt_realm_holder 领地生效

skonester_cnt_apply_max_control_effect = {
  if = {
    limit = {
      scope:actor = { has_character_flag = skonester_cnt_domain_scope }
      exists = scope:skonester_cnt_realm_holder
    }
    scope:skonester_cnt_realm_holder = {
      every_held_title = {
        limit = { tier = tier_county }
        change_county_control = 100
      }
    }
  }
  else_if = {
    limit = {
      scope:actor = { has_character_flag = skonester_cnt_vassal_scope }
      exists = scope:skonester_cnt_realm_holder
    }
    scope:skonester_cnt_realm_holder = {
      every_sub_realm_county = {
        limit = {
          holder = {
            NOT = { this = scope:skonester_cnt_realm_holder }
            is_vassal_or_below_of = scope:skonester_cnt_realm_holder
          }
        }
        change_county_control = 100
      }
    }
  }
}
