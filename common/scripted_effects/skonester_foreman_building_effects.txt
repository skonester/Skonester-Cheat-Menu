# 伯爵领：在尚无地产的地格上起一座新地产（每次至多一格）
skonester_foreman_county_place_one_empty = {
  if = {
    limit = { any_county_province = { has_holding = no } }
    if = {
      limit = { any_county_province = { has_holding = yes has_holding_type = tribal_holding } }
      random_county_province = {
        limit = { has_holding = no barony_cannot_construct_holding = no }
        set_holding_type = tribal_holding
        skonester_ensure_tribal_holding_base_tribe_building = yes
      }
    }
    else_if = {
      limit = { holder = { government_has_flag = government_is_tribal } }
      random_county_province = {
        limit = { has_holding = no barony_cannot_construct_holding = no }
        set_holding_type = tribal_holding
        skonester_ensure_tribal_holding_base_tribe_building = yes
      }
    }
    else_if = {
      limit = { holder = { government_has_flag = government_is_nomadic } }
      random_county_province = {
        limit = { has_holding = no barony_cannot_construct_holding = no }
        set_holding_type = nomad_holding
      }
    }
    else = {
      random_county_province = {
        limit = { has_holding = no barony_cannot_construct_holding = no }
        set_holding_type = castle_holding
        skonester_ensure_castle_holding_base_castle_building = yes
      }
    }
  }
}

# 分批：每脉冲处理若干县（效果与单帧版相同，仅拆开执行）
skonester_foreman_realm_batch_step_effect = {
  while = {
    count = 10
    limit = {
      variable_list_size = {
        name = skonester_foreman_realm_county_queue
        value > 0
      }
    }
    ordered_in_list = {
      variable = skonester_foreman_realm_county_queue
      order_by = skonester_foreman_batch_order_value
      position = 0
      save_scope_as = skonester_foreman_batch_county
    }
    remove_list_variable = {
      name = skonester_foreman_realm_county_queue
      target = scope:skonester_foreman_batch_county
    }
    scope:skonester_foreman_batch_county = {
      skonester_foreman_realm_one_county_effect = yes
    }
  }
}
