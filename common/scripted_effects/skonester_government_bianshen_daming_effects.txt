# 变身大明十政体（政体须由《变身大明》workshop 模组在 common/governments 中提供；本包不再带占位，避免覆盖对方完整定义导致无地产/国库收入等异常）— 批量/本人转换；游牧部落首都先改为城堡（土司/海盗改为部落）

skonester_convert_character_to_gm_zongfan_effect = {
  if = {
    limit = {
      skonester_actor_has_bianshen_daming_government_access_trigger = yes
      NOT = { has_government = zongfan_government }
    }
    if = {
      limit = { is_landed = yes }
      capital_province = {
        if = {
          limit = {
            OR = {
              has_holding_type = tribal_holding
              has_holding_type = herder_holding
              has_holding_type = nomad_holding
            }
          }
          set_holding_type = castle_holding
          skonester_ensure_castle_holding_base_castle_building = yes
        }
      }
    }
    change_government = zongfan_government
  }
}

skonester_mass_convert_vassals_to_gm_zongfan_effect = {
  every_vassal_or_below = {
    limit = {
      skonester_mass_gov_conversion_target_trigger = yes
      NOT = { has_government = zongfan_government }
    }
    skonester_convert_character_to_gm_zongfan_effect = yes
  }
}

      capital_province = {
        if = {

          }

          skonester_ensure_castle_holding_base_castle_building = yes

      tgp_domicile_conversion_when_changing_government_type = {
        NEW_DOMICILE_TYPE = east_asian_estate
        NEW_GOVERNMENT_TYPE = sansizhi_government

  }
}
