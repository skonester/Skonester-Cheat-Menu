# 将 scope:recipient 身上所有玩家尚未知晓的秘密，以「知晓」方式告知 scope:actor（不公开曝光）

skonester_learn_all_official_secrets_to_actor_effect = {
  scope:recipient = {
    every_secret = {
      limit = {
        NOT = { is_known_by = scope:actor }
      }
      reveal_to_without_events_effect = {
        CHARACTER = scope:actor
      }
    }
  }
}

      limit = {
        NOT = { is_known_by = root }
      }

        CHARACTER = root
      }
    }
  }
}
