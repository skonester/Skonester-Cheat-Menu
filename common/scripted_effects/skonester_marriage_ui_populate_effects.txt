# 强制联姻界面：将一方及其宫廷/家族加入可选列表 characters

skonester_marriage_ui_add_root_to_list_effect = {
  if = {
    limit = {
      is_alive = yes
      NOT = { is_in_list = characters }
    }
    add_to_list = characters
  }
}

skonester_marriage_ui_populate_pool_from_scope_effect = {
  skonester_marriage_ui_add_root_to_list_effect = yes
  every_courtier = {
    limit = {
      NOT = { is_in_list = characters }
    }
    add_to_list = characters
  }
  every_courtier_away = {
    limit = {
      NOT = { is_in_list = characters }
    }
    add_to_list = characters
  }
  every_child = {
    even_if_dead = yes
    limit = {
      is_alive = yes
      NOT = { is_in_list = characters }
    }
    add_to_list = characters
    every_child = {
      even_if_dead = yes
      limit = {
        is_alive = yes
        NOT = { is_in_list = characters }
      }
      add_to_list = characters
      every_child = {
        limit = {
          is_alive = yes
          NOT = { is_in_list = characters }
        }
        add_to_list = characters
      }
    }
  }
}
