skonester_positions_mass_reshuffle_liege_effect = {
  save_scope_as = skonester_pos_reshuffle_liege
  scope:skonester_pos_reshuffle_liege = {
    # --- 任免前读取：内阁成员 + 各已知宫廷职在任者写入 variable_list（skonester_pos_snap_*）---
    skonester_positions_snapshot_before_reshuffle_effect = yes

    # --- 内阁：先登记再全部解雇（避免 every_councillor 中途改表漏解雇）---
    if = {
      limit = { has_variable_list = skonester_pos_council_fire_queue }
      clear_variable_list = skonester_pos_council_fire_queue
    }
    every_councillor = {
      save_scope_as = skonester_pos_co_fire
      scope:skonester_pos_reshuffle_liege = {
        add_to_variable_list = {
          name = skonester_pos_council_fire_queue
          target = scope:skonester_pos_co_fire
        }
      }
      clear_saved_scope = skonester_pos_co_fire
    }
    if = {
      limit = {
        has_variable_list = skonester_pos_council_fire_queue
        variable_list_size = {
          name = skonester_pos_council_fire_queue
          value > 0
        }
      }
      every_in_list = {
        variable = skonester_pos_council_fire_queue
        save_scope_as = skonester_pos_co_fire
        scope:skonester_pos_reshuffle_liege = {
          fire_councillor = scope:skonester_pos_co_fire
        }
        clear_saved_scope = skonester_pos_co_fire
      }
    }
    if = {
      limit = { has_variable_list = skonester_pos_council_fire_queue }
      clear_variable_list = skonester_pos_council_fire_queue
    }

    # --- 宫廷职位：撤销与补任（完整列表见 skonester_positions_all_named_court_slots_effects.txt）---
    skonester_positions_revoke_all_named_court_positions_effect = yes

    # --- 内阁：标准五席 + 库里台四席（先严格择优，空缺宽松+生成）；TGP 天朝神州额外席同序 ---
    skonester_positions_refill_all_common_council_seats_strict_effect = yes
    skonester_positions_fill_remaining_council_loose_then_spawn_effect = yes

    # --- 宫廷职：每职严格 → 宽松 → 生成 ---
    skonester_positions_appoint_all_named_court_positions_strict_loose_spawn_effect = yes

    # --- 内阁任务：引擎默认任务 ---
    every_councillor = {
      start_default_task = yes
    }
  }
}

# 由 skonester_positions_mass_reshuffle_liege_effect 在 scope:skonester_pos_reshuffle_liege 内调用
skonester_positions_try_appoint_court_slot_effect = {
  if = {
    limit = {
      can_employ_court_position_type = $COURT_POS$
      any_courtier = {
        skonester_positions_loose_court_appointment_candidate_for_reshuffle_trigger = yes
      }
    }
    ordered_courtier = {
      limit = {
        skonester_positions_loose_court_appointment_candidate_for_reshuffle_trigger = yes
      }
      order_by = aptitude:$COURT_POS$
      max = 1
      check_range_bounds = no
      save_scope_as = skonester_pos_court_pick
    }
    if = {
      limit = { exists = scope:skonester_pos_court_pick }
      appoint_court_position = {
        recipient = scope:skonester_pos_court_pick
        court_position = $COURT_POS$
      }
      clear_saved_scope = skonester_pos_court_pick
    }
  }
  if = {
    limit = {
      can_employ_court_position_type = $COURT_POS$
      any_vassal = {
        skonester_positions_court_reshuffle_vassal_loose_candidate_trigger = yes
      }
    }
    ordered_vassal = {
      limit = {
        skonester_positions_court_reshuffle_vassal_loose_candidate_trigger = yes
      }
      order_by = aptitude:$COURT_POS$
      max = 1
      check_range_bounds = no
      save_scope_as = skonester_pos_court_pick
    }
    if = {
      limit = { exists = scope:skonester_pos_court_pick }
      appoint_court_position = {
        recipient = scope:skonester_pos_court_pick
        court_position = $COURT_POS$
      }
      clear_saved_scope = skonester_pos_court_pick
    }
  }
}
