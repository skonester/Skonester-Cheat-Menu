          appoint_court_position = {
            recipient = scope:skonester_pos_spawned
            court_position = $COURT_POS$
          }
          clear_saved_scope = skonester_pos_spawned
        }
        else_if = {
          limit = { exists = scope:skonester_pos_spawned }
          clear_saved_scope = skonester_pos_spawned
        }
      }
      change_local_variable = {
        name = skonester_pos_slot_fill_iter
        add = 1
      }
    }
  }
