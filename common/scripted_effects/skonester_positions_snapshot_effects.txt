# 若该职在任，把在任者加入 $LIST_NAME$（每职至多一人）
skonester_positions_snapshot_one_court_holder_effect = {
  if = {
    limit = { exists = scope:skonester_pos_snap_tmp_holder }
    clear_saved_scope = skonester_pos_snap_tmp_holder
  }
  scope:skonester_pos_reshuffle_liege = {
    if = {
      limit = { employs_court_position = $COURT_POS$ }
      random_court_position_holder = {
        type = $COURT_POS$
        limit = { always = yes }
        save_scope_as = skonester_pos_snap_tmp_holder
      }
    }
    if = {
      limit = { exists = scope:skonester_pos_snap_tmp_holder }
      add_to_variable_list = {
        name = $LIST_NAME$
        target = scope:skonester_pos_snap_tmp_holder
      }
      clear_saved_scope = skonester_pos_snap_tmp_holder
    }
  }
}
