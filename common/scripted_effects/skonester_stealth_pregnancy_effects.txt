# 隐秘怀孕效果 - 隐藏玩家父亲身份
# 根据女性婚姻状况设置不同的假定父亲

skonester_stealth_force_pregnancy_effect = {
  scope:recipient = {
    # 检查女性的婚姻状况
    if = {
      limit = {
        # 已婚且有主配偶
        exists = primary_spouse
      }
      # 设置主配偶为假定父亲
      make_pregnant = {
        father = scope:actor
        known_bastard = no
      }
      primary_spouse = {
        save_scope_as = skonester_stealth_assumed_father
      }
      set_pregnancy_assumed_father = scope:skonester_stealth_assumed_father
      clear_saved_scope = skonester_stealth_assumed_father
    }
    else_if = {
      limit = {
        # 检查是否是侧室或小妾
        has_trait = concubine
        exists = liege
      }
      # 设置领主为假定父亲
      make_pregnant = {
        father = scope:actor
        known_bastard = no
      }
      liege = {
        save_scope_as = skonester_stealth_assumed_father
      }
      set_pregnancy_assumed_father = scope:skonester_stealth_assumed_father
      clear_saved_scope = skonester_stealth_assumed_father
    }
    else_if = {
      limit = {
        # 检查是否是情人
        any_relation = { type = lover }
      }
      # 设置情人为假定父亲
      make_pregnant = {
        father = scope:actor
        known_bastard = no
      }
      random_relation = {
        type = lover
        save_scope_as = skonester_stealth_assumed_father
      }
      set_pregnancy_assumed_father = scope:skonester_stealth_assumed_father
      clear_saved_scope = skonester_stealth_assumed_father
    }
    else = {
      # 未婚女性 - 不设置假定父亲，只显示母亲
      make_pregnant = {
        father = scope:actor
        known_bastard = no
      }
      # 不设置假定父亲，让孩子只显示母亲
    }
  }
}
