skonester_treasure_maker_apply_god_packs_from_actor_effect = {
  if = {
    limit = { scope:actor = { has_character_flag = skonester_tm_god_longlife } }
    add_artifact_modifier = skonester_tm_godpack_vitality
  }
  if = {
    limit = { scope:actor = { has_character_flag = skonester_tm_god_youth } }
    add_artifact_modifier = skonester_tm_godpack_youth
  }
  if = {
    limit = { scope:actor = { has_character_flag = skonester_tm_god_army_mass } }
    add_artifact_modifier = skonester_tm_godpack_army
  }
  if = {
    limit = { scope:actor = { has_character_flag = skonester_tm_god_scheme_kill } }
    add_artifact_modifier = skonester_tm_godpack_scheme_attack
  }
  if = {
    limit = { scope:actor = { has_character_flag = skonester_tm_god_scheme_immune } }
    add_artifact_modifier = skonester_tm_godpack_scheme_defense
  }
  if = {
    limit = { scope:actor = { has_character_flag = skonester_tm_god_genius_children } }
    add_artifact_modifier = skonester_tm_godpack_dynasty
  }
  if = {
    limit = { scope:actor = { has_character_flag = skonester_tm_god_tax } }
    add_artifact_modifier = skonester_tm_godpack_tax
  }
  if = {
    limit = { scope:actor = { has_character_flag = skonester_tm_god_vassal_calm } }
    add_artifact_modifier = skonester_tm_godpack_vassal
  }
  if = {
    limit = { scope:actor = { has_character_flag = skonester_tm_god_faith_culture } }
    add_artifact_modifier = skonester_tm_godpack_faith_culture
  }
  if = {
    limit = { scope:actor = { has_character_flag = skonester_tm_god_dynasty } }
    add_artifact_modifier = skonester_tm_godpack_dread_piety
  }
  if = {
    limit = { scope:actor = { has_character_flag = skonester_tm_god_knights } }
    add_artifact_modifier = skonester_tm_godpack_knight_army
  }
  if = {
    limit = {
      scope:actor = {
        OR = {
          has_character_flag = skonester_tm_god_longlife
          has_character_flag = skonester_tm_god_army_mass
        }
      }
    }
    add_artifact_modifier = skonester_tm_godpack_mobility
  }
}

skonester_treasure_maker_spawn_for_recipient_effect = {
  if = {
    limit = {
      exists = scope:actor
      scope:actor = { skonester_player_may_use_artifact_craft_trigger = yes }
    }
  scope:actor = {
    if = {
      limit = { has_character_flag = skonester_tm_kind_armor }
      if = {
        limit = { has_character_flag = skonester_tm_quality_low }
        create_artifact = {
          name = skonester_tm_artifact_name
          description = skonester_tm_artifact_desc
          creator = scope:actor
          template = unique_artifact_template
          type = armor_plate
          visuals = armor
          wealth = 20
          quality = 20
          decaying = no
          modifier = artifact_placeholder_modifier
          save_scope_as = skonester_tm_new_art
