skonester_acm_gui_close = {
  scope = character
  is_shown = { always = yes }
  is_valid = { always = yes }
  effect = {
    remove_character_flag = skonester_acm_gui_open
    remove_character_flag = skonester_acm_modify_session
    remove_character_flag = skonester_acm_repair_gui_open
    remove_character_flag = skonester_acm_edit_from_craft_gui
    remove_character_flag = skonester_acm_edit_from_repair_gui
    remove_character_flag = skonester_acm_gui_text_edit_active
    if = {
      limit = { has_global_variable = skonester_acm_recipient }
      remove_global_variable = skonester_acm_recipient
    }
    if = {
      limit = { has_global_variable = skonester_acm_repair_target }
      remove_global_variable = skonester_acm_repair_target
    }
    if = {
      limit = { has_variable = skonester_acm_repair_artifact_var }
      remove_variable = skonester_acm_repair_artifact_var
    }
  }
}
