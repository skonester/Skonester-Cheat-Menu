skonester_console_helper_open_visible = {
  scope = character
  is_shown = { has_character_flag = skonester_console_helper_open }
  is_valid = { always = yes }
}

skonester_console_helper_close = {
  scope = character
  is_shown = { always = yes }
  is_valid = { always = yes }
  effect = {
    remove_character_flag = skonester_console_helper_open
  }
}
