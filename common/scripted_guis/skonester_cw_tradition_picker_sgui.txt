skonester_cw_tradition_category_list_maker = {
  scope = character

  saved_scopes = { tradition }

  effect = {
    if = {
      limit = {
        scope:tradition = {
          has_tradition_category = realm
        }
      }

      add_to_global_variable_list = { name = skonester_cw_realm_tradition_list target = scope:tradition}
    }
    else_if = {
      limit = {
        scope:tradition = {
          has_tradition_category = combat
        }
      }

      add_to_global_variable_list = { name = skonester_cw_combat_tradition_list target = scope:tradition}
    }
    else_if = {
      limit = {
        scope:tradition = {
          has_tradition_category = societal
        }
      }

      add_to_global_variable_list = { name = skonester_cw_societal_tradition_list target = scope:tradition}
    }
    else_if = {
      limit = {
        scope:tradition = {
          has_tradition_category = ritual
        }
      }

      add_to_global_variable_list = { name = skonester_cw_ritual_tradition_list target = scope:tradition}
    }
    else_if = {
      limit = {
        scope:tradition = {
          has_tradition_category = regional
        }
      }

      add_to_global_variable_list = { name = skonester_cw_regional_tradition_list target = scope:tradition}
    }
  }
}

skonester_cw_add_tradition = {
  scope = character

  saved_scopes = { tradition }

  effect = {
    var:skonester_selected_culture_var = {
      add_culture_tradition = scope:tradition
    }

  }
}

skonester_cw_remove_tradition = {
  scope = character

  saved_scopes = { tradition }

  effect = {
    var:skonester_selected_culture_var = {
      remove_culture_tradition = scope:tradition
    }

  }
}

skonester_cw_culture_has_tradition = {

  scope = character

  saved_scopes = { tradition }

  is_valid = {
    var:skonester_selected_culture_var = { has_cultural_tradition = scope:tradition }
  }
}

skonester_cw_tradition_picker_visible = {
  scope = character
  is_shown = { has_character_flag = skonester_cw_tradition_picker_open }
}

skonester_cw_open_tradition_picker_window = {
  scope = character
  is_shown = { is_ai = no }
  is_valid = {
    exists = var:skonester_selected_culture_var
  }
}

skonester_cw_close_tradition_picker_window = {
  scope = character
  is_valid = { always = yes }
  effect = {
    remove_character_flag = skonester_cw_tradition_picker_open
    skonester_cw_clear_tradition_picker_lists_effect = yes
  }
}
