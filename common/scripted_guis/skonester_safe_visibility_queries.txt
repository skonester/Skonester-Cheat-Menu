# Rootless visibility queries keep the original character conditions behind an optional named player.

skonester_lsf_gui_visible_safe_query = {
  # No character root: the frontend can ask before a player exists.
  is_shown = {
    exists = scope:skonester_gui_player
    scope:skonester_gui_player ?= { has_character_flag = skonester_lsf_gui_open 
    }
  }
}
