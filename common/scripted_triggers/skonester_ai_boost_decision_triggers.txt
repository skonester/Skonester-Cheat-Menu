# 相对邻国：己方 max 兵力 ≥ 对方 × 0.9（微弱优势 / 均势偏强）
skonester_ai_boost_neighbor_rel_power_090_trigger = {
  any_neighboring_top_liege_realm_owner = {
    root.max_military_strength >= {
      value = this.max_military_strength
      multiply = 0.9
    }
  }
}

# 直辖/封臣领地内：与统治者文化不同的「异文化种类」计数（glob 文化 + prev 绑定）

skonester_ai_boost_foreign_culture_count_equals_1_trigger = {

    count = 1

      }
    }
  }
}

# 任一独立游玩的人类统治者 realm_size > 30（全局「大户玩家」加压）
skonester_ai_boost_any_player_realm_gt_30_trigger = {
  any_living_character = {
    is_ai = no
    is_playable_character = yes
    is_independent_ruler = yes
    realm_size > 30
  }
}

# root 与人类独立统治者首都同属 de jure empire 或 kingdom（近似「与玩家同城际宏观区」，原版无 same_region）
skonester_ai_boost_same_macro_region_as_any_player_trigger = {
  any_living_character = {
    is_ai = no
    is_playable_character = yes
    is_independent_ruler = yes
    OR = {
      capital_county.empire ?= root.capital_county.empire
      capital_county.kingdom ?= root.capital_county.kingdom
    }
  }
}

# 「吞地」：任一人类独立 ruler realm≥55 / realm≥38 且在战 / 征服者（加压全域 AI）
skonester_ai_boost_player_blobbing_trigger = {
  any_living_character = {
    is_ai = no
    is_playable_character = yes
    is_independent_ruler = yes
    OR = {
      realm_size >= 55
      AND = {
        realm_size >= 38
        is_at_war = yes
      }
      has_trait = conqueror
    }
  }
}
