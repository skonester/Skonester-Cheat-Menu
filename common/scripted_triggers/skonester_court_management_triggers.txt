# 管理廷臣：受保護角色、惡質特質、招募候選門檻（this = 廷臣或候選人；root = 玩家）
# 遣散保護收窄：僅父母、同胞、配偶、妾、與玩家有法親/血親之子女；同宗族遠親可遣散。
# 另保護：人質；玩家為其監護人之未成年；與玩家有 guardian 腳本關係者（事件常見）。

skonester_court_mgmt_protected_trigger = {
  OR = {
    this = root
    is_spouse_of = root
    is_concubine_of = root
    is_parent_of = root
    is_sibling_of = root
    AND = {
      is_child_of = root
      OR = {
        father = root
        mother = root
      }
    }
    is_hostage = yes
    has_relation_guardian = root
    has_secret_relation_guardian = root
  }
}

# 惡劣性格（與一鍵監護人對齊並擴充：偏執、暴食、嗜酒、刻薄、報復、善妒等）

}

# 人數較多時優先保留：頂級教育／勇武教育直接保留；其餘須天才系 + 五維均達標
skonester_court_mgmt_elite_keep_trigger = {
  OR = {
    skonester_court_mgmt_signature_star_trigger = yes
    AND = {
      OR = {
        has_trait = intellect_good_3
        AND = {
          has_trait = intellect_good_2
          learning >= 18
        }
        AND = {
          has_trait = shrewd
          learning >= 20
        }
      }
      martial >= 16
      diplomacy >= 16
      stewardship >= 16
      intrigue >= 16
      learning >= 16
    }
  }
}

# 人數偏高：遣散「無智力線特質且總能力偏低」者（頂級教育／勇武教育招牌不觸發此條）
skonester_court_mgmt_soft_cull_trigger = {
  NOT = {
    skonester_court_mgmt_signature_star_trigger = yes
  }
  NOR = {
    has_trait = intellect_good_3
    has_trait = intellect_good_2
    has_trait = shrewd
  }
  sum_of_all_skills_value < 95
}

skonester_court_mgmt_purge_candidate_trigger = {
  NOT = {
    skonester_court_mgmt_protected_trigger = yes
  }
  OR = {
    skonester_court_mgmt_barred_trait_trigger = yes
    skonester_court_mgmt_three_main_skills_under_10_trigger = yes
    AND = {
      root = {
        var:skonester_court_n > 45
      }
      NOT = {
        skonester_court_mgmt_elite_keep_trigger = yes
      }
    }
    AND = {
      root = {
        var:skonester_court_n > 38
      }
      skonester_court_mgmt_soft_cull_trigger = yes
    }
  }
}

# 廷臣硬上限：超過定額時可遣散的非受保護廷臣（與 purge 優先序相同）
skonester_court_mgmt_cap_overflow_candidate_trigger = {
  is_courtier_of = root
  NOT = {
    skonester_court_mgmt_protected_trigger = yes
  }
}
