# 疆域内是否存在可改为城堡封建地产的男爵领：部落 / 游牧 / 牧民（不含已是城堡）

skonester_has_tribal_holding_in_realm_trigger = {
  any_sub_realm_county = {
    any_county_province = {
      OR = {
        has_holding_type = tribal_holding
        has_holding_type = nomad_holding
        has_holding_type = herder_holding
      }
    }
  }
}
