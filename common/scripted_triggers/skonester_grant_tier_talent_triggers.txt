# 多余头衔授予：拟授头衔越高，文/武能力线越高（见 skonester_purge_governor_skill_values.txt）

skonester_is_talent_for_grant_tier_barony_trigger = {
  OR = {
    AND = {
      skonester_is_gm_yan_military_government_trigger = yes
      skonester_military_skill_sum_for_purge_effective_value >= skonester_military_grant_barony_threshold_value
    }
    AND = {
      skonester_is_gm_yan_civil_government_trigger = yes
      skonester_civil_skill_sum_for_purge_effective_value >= skonester_civil_grant_barony_threshold_value
    }
    AND = {
      NOT = { skonester_is_gm_yan_government_trigger = yes }
      OR = {
        vassal_contract_has_flag = celestial_military_appointment
        vassal_contract_has_flag = meritocratic_military_appointment
      }
      skonester_military_skill_sum_for_purge_value >= skonester_military_grant_barony_threshold_value
    }
    AND = {
      NOT = { skonester_is_gm_yan_government_trigger = yes }
      NOR = {
        vassal_contract_has_flag = celestial_military_appointment
        vassal_contract_has_flag = meritocratic_military_appointment
      }
      is_landed = yes
      skonester_civil_skill_sum_for_purge_effective_value >= skonester_civil_grant_barony_threshold_value
    }
    AND = {
      NOT = { is_landed = yes }
      OR = {
        skonester_civil_skill_sum_for_purge_value >= skonester_civil_grant_barony_threshold_value
        skonester_military_skill_sum_for_purge_value >= skonester_military_grant_barony_threshold_value
      }
    }
  }
}
