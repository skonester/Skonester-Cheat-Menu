# 隐秘怀孕触发器 - 用于隐藏玩家父亲身份
# 根据女性婚姻状况决定是否隐藏父亲身份

# 检查是否需要隐藏玩家父亲身份
# 已婚 / 侧室 / 未婚三种情况下原逻辑都是"隐藏"，故恒真
skonester_stealth_hide_father_trigger = {
  always = yes
}

# 检查女性是否有合法的假定父亲：已婚，或为侧室且有领主，或有情人关系
skonester_stealth_has_assumed_father_trigger = {
  OR = {
    exists = primary_spouse
    AND = {
      has_trait = concubine
      exists = liege
    }
    any_relation = { type = lover }
  }
}

# 获取假定父亲的作用域（这是效果，非触发器——用 scripted_effect 显式声明类型）
scripted_effect skonester_stealth_get_assumed_father_trigger = {
  # 已婚 - 返回主配偶
  if = {
    limit = {
      exists = primary_spouse
    }
    primary_spouse = {
      save_scope_as = skonester_stealth_assumed_father
    }
  }
  # 侧室/小妾 - 返回领主
  else_if = {
    limit = {
      has_trait = concubine
      exists = liege
    }
    liege = {
      save_scope_as = skonester_stealth_assumed_father
    }
  }
  # 情人 - 返回情人
  else_if = {
    limit = {
      any_relation = { type = lover }
    }
    random_relation = {
      type = lover
      save_scope_as = skonester_stealth_assumed_father
    }
  }
}
