# 法理链上无境内持有者：强制直属 root（仍在 root 版图内）
skonester_land_title_de_jure_fallback_root_trigger = {
  exists = holder
  NOT = { holder = root }
  NOT = { holder.liege = root }
  holder = {
    is_vassal_or_below_of = root
    is_landed = yes
    NOT = { is_at_war_with = root }
  }
  NOR = {
    skonester_land_title_de_jure_layer_1_needs_fix_trigger = yes
    skonester_land_title_de_jure_layer_2_needs_fix_trigger = yes
    skonester_land_title_de_jure_layer_3_needs_fix_trigger = yes
    skonester_land_title_de_jure_layer_4_needs_fix_trigger = yes
    skonester_land_title_de_jure_layer_5_needs_fix_trigger = yes
    skonester_land_title_de_jure_layer_6_needs_fix_trigger = yes
  }
  OR = {
    skonester_land_title_de_jure_layer_1_resolved_trigger = yes
    skonester_land_title_de_jure_layer_2_resolved_trigger = yes
    skonester_land_title_de_jure_layer_3_resolved_trigger = yes
    skonester_land_title_de_jure_layer_4_resolved_trigger = yes
    skonester_land_title_de_jure_layer_5_resolved_trigger = yes
    skonester_land_title_de_jure_layer_6_resolved_trigger = yes
    NOT = {
      holder.liege = {
        skonester_de_jure_transfer_valid_liege_in_realm_trigger = yes
      }
    }
  }
}
