  type = character_event
  title = skonester_acm_edit_name_evt_t
  desc = skonester_acm_edit_name_evt_d
  theme = stewardship
  left_portrait = root

  widget = {
    gui = "event_window_widget_enter_text"
    container = "dynamic_birth_name"
    controller = {
      type = text
      data = {
        key = skonester_acm_edit_artifact_name
        default = skonester_acm_edit_name_default
      }
    }
    setup_scope = {
      root = { save_scope_as = text_target }
    }
  }

  option = {
    name = skonester_acm_custom_text_confirm
    root = {
      add_character_flag = skonester_acm_edit_name_confirmed
      skonester_acm_refresh_artifact_text_preview_effect = yes
      skonester_acm_reopen_artifact_text_editor_gui_effect = yes
    }
  }

  option = {
    name = skonester_acm_custom_text_clear
    root = {
      skonester_acm_clear_custom_artifact_name_effect = yes
      skonester_acm_reopen_artifact_text_editor_gui_effect = yes
    }
  }

  option = {
    name = skonester_cheat_menu.00x.close
    root = {
      skonester_acm_reopen_artifact_text_editor_gui_effect = yes
    }
  }
}
