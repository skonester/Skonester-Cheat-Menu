namespace = skonester_cheat_menu

# 伯爵领作弊：以右键角色为「领地统治者」作用域；先选其直辖/封臣伯爵领，再选四项拉满（控制力、发展度、民众好感、文化信仰）

skonester_cheat_menu.3070 = {
  type = character_event
  title = skonester_cnt.3070.t
  desc = skonester_ui_no_hover_tooltip
  theme = stewardship
  left_portrait = scope:actor
  right_portrait = scope:skonester_cnt_edit_target

  immediate = {
    if = {
      limit = { exists = scope:skonester_cnt_edit_target }
      scope:skonester_cnt_edit_target = { save_scope_as = skonester_cnt_realm_holder }
    }
    else = {
      scope:actor = { save_scope_as = skonester_cnt_realm_holder }
    }
  }

  option = {
    name = skonester_cnt.3070.pick_domain
    trigger = {
      skonester_cheat_menu_menu_enabled_trigger = yes
      scope:skonester_cnt_realm_holder = {
        is_playable_character = yes
        is_landed = yes
      }
    }
    scope:actor = {
      remove_character_flag = skonester_cnt_vassal_scope
      add_character_flag = skonester_cnt_domain_scope
      trigger_event = skonester_cheat_menu.3075
    }
  }
  option = {
    name = skonester_cnt.3070.pick_vassal_counties
    trigger = {
      skonester_cheat_menu_menu_enabled_trigger = yes
      scope:skonester_cnt_realm_holder = {
        is_playable_character = yes
        is_landed = yes
      }
    }
    scope:actor = {
      remove_character_flag = skonester_cnt_domain_scope
      add_character_flag = skonester_cnt_vassal_scope
      trigger_event = skonester_cheat_menu.3075
    }
  }
  option = {
    name = skonester_cheat_menu.00x.close
    scope:actor = {
      remove_character_flag = skonester_cnt_domain_scope
      remove_character_flag = skonester_cnt_vassal_scope
      if = {
        limit = { has_variable = skonester_cnt_edit_target }
        remove_variable = skonester_cnt_edit_target
      }
    }
  }
}

skonester_cheat_menu.3075 = {

  title = skonester_cnt.3075.t
  desc = skonester_ui_no_hover_tooltip
  theme = stewardship
  left_portrait = scope:actor
  right_portrait = scope:skonester_cnt_edit_target

  immediate = {
    if = {
      limit = { NOT = { exists = scope:skonester_cnt_realm_holder } }
      if = {
        limit = { exists = scope:skonester_cnt_edit_target }
        scope:skonester_cnt_edit_target = { save_scope_as = skonester_cnt_realm_holder }

      else = {
        scope:actor = { save_scope_as = skonester_cnt_realm_holder }

    name = skonester_cnt.3075.max_control

      skonester_cheat_menu_menu_enabled_trigger = yes

        OR = {
          has_character_flag = skonester_cnt_domain_scope
          has_character_flag = skonester_cnt_vassal_scope
        }
      }

    scope:actor = {

    skonester_cnt_edit_target } }
