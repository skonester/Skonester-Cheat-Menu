# 战争中每 7 天拉满军团补给（仅玩家且已开启军需修正）
skonester_cheat_menu.3099 = {
  type = character_event
  hidden = yes
  immediate = {
    if = {
      limit = {
        is_alive = yes
        has_character_modifier = skonester_military_instant_replenish_modifier
      }
      skonester_cheat_menu_refill_all_army_supply_on_character_effect = yes
      if = {
        limit = { is_at_war = yes }
        trigger_event = { id = skonester_cheat_menu.3099 days = 7 }
      }
    }
  }
}
