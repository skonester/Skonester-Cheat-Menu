# 厉鬼勾魂：处决波次（手点「开启」可立即触发，或由 yearly 满 5 年时触发）
# 仅你顶层领国（与 is_in_same_realm_as=root 一致）；死因为随机疫病；人类玩家对每名死者发界面提示条（与瘟疫式提示类似）。清全局开关/年计。
skonester_officials.9086 = {
  type = character_event
  hidden = yes

  immediate = {
    root = {
      set_variable = {
        name = skonester_samge_left
        value = 1000
      }
      set_variable = {
        name = skonester_samge_guard
        value = 0
      }
    }
    while = {
      limit = {
        root = {
          var:skonester_samge_left > 0
          var:skonester_samge_guard < 12000
        }
      }
      root = {
        change_variable = {
          name = skonester_samge_guard
          add = 1
        }
      }
      if = {
        limit = {
          any_living_character = {
            is_ai = yes
            is_alive = yes
            NOT = { this = root }
            top_liege = root.top_liege
          }
        }
        random_living_character = {
          limit = {
            is_ai = yes
            is_alive = yes
            NOT = { this = root }
            top_liege = root.top_liege
          }
          weight = {
            base = 1
          }
          save_scope_as = skonester_samge_victim
          root = {
            if = {
              limit = { is_ai = no }
              send_interface_toast = {
                type = event_toast_effect_neutral
                title = skonester_samge_realm_death_toast_t
                left_icon = scope:skonester_samge_victim
                custom_description_no_bullet = {
                  object = scope:skonester_samge_victim
                  text = skonester_samge_realm_death_toast_desc
                }
              }
            }
          }
          random_list = {
            25 = {
              death = {
                death_reason = death_great_pox
              }
            }
            25 = {
              death = {
                death_reason = death_typhus
              }
            }
            25 = {
              death = {
                death_reason = death_smallpox
              }
            }
            25 = {
              death = {
                death_reason = death_measles
              }
            }
          }
        }
        root = {
          change_variable = {
            name = skonester_samge_left
            add = -1
          }
        }
      }
      else = {
        root = {
          set_variable = {
            name = skonester_samge_left
            value = 0
          }
        }
      }
    }
    root = {
      if = {
        limit = {
          has_variable = skonester_samge_left
        }
        remove_variable = skonester_samge_left
      }
      if = {
        limit = {
          has_variable = skonester_samge_guard
        }
        remove_variable = skonester_samge_guard
      }
    }
    # 一轮结束即关「开」、清年计（新逻辑点按一次一轮；旧存档年循环在开关仍亮时由 yearly 触发的也一并收尾）
    if = {
      limit = {
        has_global_variable = skonester_samge_reaping_active
      }
      remove_global_variable = skonester_samge_reaping_active
    }
    if = {
      limit = {
        has_global_variable = skonester_samge_years_accum
      }
      remove_global_variable = skonester_samge_years_accum
    }
    if = {
      limit = {
        root = {
          has_character_flag = skonester_samge_reopen_pop_9080_after_samge_wave
        }
      }
      root = {
        remove_character_flag = skonester_samge_reopen_pop_9080_after_samge_wave
        trigger_event = {
          id = skonester_officials.9080
          days = 0
        }
      }
    }
  }
}
