	option = { name = skonester_quick_modifications.1000.editors trigger_event = skonester_quick_modifications.1200 }
	option = { name = skonester_quick_modifications.1000.modifiers trigger_event = skonester_quick_modifications.1300 }
	option = { name = skonester_quick_modifications.1000.realm trigger_event = skonester_quick_modifications.1400 }
	option = {
		name = skonester_quick_modifications.1000.innovations
		trigger = { scope:skonester_cheat_menu_target = { exists = culture } }
		scope:skonester_cheat_menu_target.culture = { save_scope_as = skonester_selected_culture }
		scope:actor = {
			remove_character_flag = skonester_culture_innov_from_0120

			add_character_flag = skonester_quick_modifications_innovation_entry
			trigger_event = skonester_cheat_menu.0130
		}
	}
	option = { name = skonester_quick_modifications.close }
}
