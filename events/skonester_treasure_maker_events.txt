skonester_treasure_maker.0001 = {
  type = character_event
  title = skonester_treasure_maker.0001.t
  desc = skonester_treasure_maker.0001.desc
  theme = stewardship
  right_portrait = scope:actor

  option = {
    name = skonester_treasure_maker.0001.owner_self
    trigger = {
      exists = scope:recipient
      NOT = { scope:recipient = { this = scope:actor } }
    }
    scope:actor = {
      remove_character_flag = skonester_tm_give_recipient
      trigger_event = skonester_treasure_maker.0001
    }
  }

  option = {
    name = skonester_treasure_maker.0001.owner_recipient
    trigger = {
      exists = scope:recipient

    scope:actor = {
      add_character_flag = skonester_tm_give_recipient

    }

  option = {
