namespace = zhuge_bloodline_god

# ───────────────────────────────────────────
# 父亲拥有 3 级血脉 → 子嗣 50% 获 3，50% 获 2
# ───────────────────────────────────────────
zhuge_bloodline_god.0001 = {
    type = character_event
    hidden = yes          # 不弹窗 (CK3 1.20: was hide_window)
    trigger = {
        # CK3 1.20: is_baby is not a trigger, and requiring is_male and is_female together meant this never fired.
        # The bloodline traits pass only to sons (child_inheritance_sex / valid_sex = male).
        age < 1
        is_male = yes
        father = { has_trait = bloodline_god_3 }
        NOT = { has_trait = bloodline_god_3 }  # 避免重复赋值
    }
    immediate = {
        random_list = {
            50 = { add_trait = bloodline_god_3 }
            50 = { add_trait = bloodline_god_2 }
        }
    }
}

# ───────────────────────────────────────────
# 父亲拥有 2 级血脉 → 子嗣 50% 获 2，50% 获 1
# ───────────────────────────────────────────
zhuge_bloodline_god.0002 = {
    type = character_event
    hidden = yes          # CK3 1.20: was hide_window
    trigger = {
        # CK3 1.20: is_baby is not a trigger, and requiring is_male and is_female together meant this never fired.
        # The bloodline traits pass only to sons (child_inheritance_sex / valid_sex = male).
        age < 1
        is_male = yes
        father = { has_trait = bloodline_god_2 }
        NOR = {  # CK3 1.20: was NOT, which only skipped characters holding both traits
            has_trait = bloodline_god_3   # 前面事件可能已赋
            has_trait = bloodline_god_2
        }
    }
    immediate = {
        random_list = {
            50 = { add_trait = bloodline_god_2 }
            50 = { add_trait = bloodline_god_1 }
        }
    }
}
